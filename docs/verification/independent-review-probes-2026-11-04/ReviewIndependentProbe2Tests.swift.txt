import XCTest
@testable import PocketCore

final class ReviewIndependentProbe2Tests: XCTestCase {
    private func makeA() throws -> (Assistant, FakePhoneEffects) {
        let e = ScriptedEngine()
        e.script(whenPromptContains: "long-term memory", logits: ["A": -3, "B": -3, "C": 3])
        e.script(whenPromptContains: "How should", logits: ["A": 5, "B": -3, "C": -3, "D": -3])
        let db = try SQLiteDB(path: ":memory:")
        let effects = FakePhoneEffects()
        let dana = candidate("Dana", id: "dana", methods: [phone("mobile", "5550103")])
        for k in ["Dana", "dana"] { effects.contactQueries[k] = ContactLookup(access: .full, matches: [dana]) }
        effects.contactIDs["dana"] = ContactLookup(access: .full, matches: [dana])
        let a = Assistant(reflex: Reflex(engine: e, telemetry: try TelemetryLog(db: db), source: "test"),
                          memory: try MemoryStore(db: db), actions: RecordingActions(),
                          ledger: try EffectLedger(db: db), effects: effects, cloud: CloudBrainChain([]), isOnline: { false })
        a.skills = SkillRegistry()
        return (a, effects)
    }

    func testProbeMessageVariants() async throws {
        let utterances = ["Send that to Dana as an email", "Text Dana about that email", "email Dana that the group chat moved",
                          "send Dana an email that says hi", "Send an email to Dana that I'm late", "text Dana that I will email you",
                          "message Dana that", "Send that to Dana", "text the group chat that dinner is at 7",
                          "text Dana saying that email is down"]
        for u in utterances {
            let (a, effects) = try makeA()
            let turn = try await a.handle(u)
            var out = "\(turn.reply)"
            if case .needsConfirmation(let c) = turn.reply { out = "CARD \(c.capabilityID) body=\(c.fields.first { $0.name == "body" }?.value ?? "-")" }
            if case .clarify(let q) = turn.reply { out = "CLARIFY \(q.prefix(90))" }
            if case .local(let q) = turn.reply { out = "LOCAL \(q.prefix(60))" }
            print("PROBE-MSG [\(u)] ->", out, "performs=\(effects.performCount)")
        }
    }

    /// The pre-85ffccc chart pairing, copied verbatim, for before/after comparison.
    private func oldPairs(_ text: String) -> [(String, Double)] {
        let numRe = try! NSRegularExpression(pattern: #"-?\$?\d[\d,]*(?:\.\d+)?%?"#)
        var out: [(String, Double)] = []
        for seg in text.components(separatedBy: CharacterSet(charactersIn: ";\n|")).flatMap({ $0.components(separatedBy: ", ") }) {
            let ns = seg as NSString
            guard let m = numRe.matches(in: seg, range: NSRange(location: 0, length: ns.length)).last else { continue }
            let raw = ns.substring(with: m.range).filter { $0.isNumber || $0 == "." || $0 == "-" }
            guard let v = Double(raw), v.isFinite else { continue }
            var label = ns.substring(to: m.range.location)
            if let c = label.lastIndex(where: { $0 == ":" || $0 == "=" }) {
                let after = String(label[label.index(after: c)...])
                label = after.contains(where: \.isLetter) ? after : String(label[..<c])
            }
            let words = label.split { !$0.isLetter && !$0.isNumber && $0 != "'" && $0 != "&" }.suffix(3)
            out.append((words.joined(separator: " "), v))
        }
        return out.enumerated().map { ($0.element.0.isEmpty ? "#\($0.offset + 1)" : $0.element.0, $0.element.1) }
    }

    func testProbeOldVersusNewChart() {
        for s in ["5 apples, 3 pears", "sales 2023 100, sales 2024 200", "Jan 2023 100, Feb 2023 200", "2 cats, 3 dogs",
                  "temperature 72 degrees, humidity 40 percent", "I paid 12 for coffee and 30 for lunch"] {
            print("PROBE-OLD [\(s)] old:", oldPairs(s).map { "\($0.0)=\($0.1)" }.joined(separator: " | "),
                  " new:", HTMLViz.pairs(from: s).map { "\($0.label)=\($0.value)" }.joined(separator: " | "))
        }
    }

    func testProbeChartEndToEndWithYears() async throws {
        let (a, _) = try makeA()
        let turn = try await a.handle("Make a chart: sales 2023 100, sales 2024 200")
        if case .local(let r) = turn.reply { print("PROBE-CHART-E2E bars:", r.prefix(40), r.contains("4 bars")) } else { print("PROBE-CHART-E2E", turn.reply) }
    }
}
